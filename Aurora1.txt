0.1 — Definición filosófica del lenguaje

text

Principios fundamentales:
├── Todo en español (palabras reservadas, errores, documentación)
├── Control de hardware como C/C++
├── Seguridad de memoria como Rust (ownership + borrowing)
├── Sintaxis limpia como Python
├── OOP completa (herencia, polimorfismo, interfaces)
├── Compilado a código nativo
├── Multiplataforma (Windows, macOS, Linux)
└── Ecosistema de librerías integrado

Ejemplo de sintaxis objetivo:

aura

importar ia.generativa desde "aura/ia"
importar ui desde "aura/interfaz"

clase Animal {
    protegido cadena nombre
    protegido entero edad

    publico funcion crear(cadena nombre, entero edad) {
        este.nombre = nombre
        este.edad = edad
    }

    publico funcion hablar() -> cadena {
        retornar "..."
    }
}

clase Perro hereda Animal {
    privado cadena raza

    publico funcion crear(cadena nombre, entero edad, cadena raza) {
        super.crear(nombre, edad)
        este.raza = raza
    }

    publico funcion hablar() -> cadena {
        retornar "¡Guau!"
    }
}

funcion principal() -> vacio {
    variable miPerro = nuevo Perro("Max", 3, "Labrador")
    variable sonido = miPerro.hablar()

    si sonido es cadena {
        imprimir("El perro dice: {sonido}")
    }

    // Manejo de memoria seguro
    variable prestamo ref = prestamo miPerro
    imprimir(ref.nombre)

    // Manejo inseguro para hardware
    inseguro {
        variable apuntador<entero> ptr = apuntador(0xFF00)
        ptr.escribir(42)
    }
}
